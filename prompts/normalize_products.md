Критерии отказа (Стоп-сигналы). 
Если во входном тексте:

Действие при успехе: Если текст прошел проверку и содержит явные наименования еды, выдели все продукты из этого текста и выведи их в виде маркированного списка (каждый продукт с новой строки через «-»).

import re

def is_valid_food_query(text):
    # 1. Проверка на белиберду (меньше 3 букв или только цифры/знаки)
    if len(re.findall(r'[а-яa-z]', text.lower())) < 3:
        return False
    
    # 2. Список стоп-слов (не еда)
    stop_words = ['порох', 'стерка', 'экран', 'дрель', 'бензин', 'клей', 'бумага']
    if any(word in text.lower() for word in stop_words):
        return False
    
    # 3. Словарь продуктов (можно расширять)
    food_keywords = ['мясо', 'рыба', 'рис', 'картошк', 'помидор', 'лук', 'морковь', 
                    'курица', 'яйцо', 'мука', 'сахар', 'соль', 'масло', 'сыр', 
                    'молоко', 'сливки', 'творог', 'гречка', 'макароны']
    
    # Если есть хоть одно съедобное слово — пропускаем
    if any(word in text.lower() for word in food_keywords):
        return True
    
    # 4. Если ничего не найдено — блокируем
    return False
